using System;
using System.Text.Json;

Console.OutputEncoding = System.Text.Encoding.UTF8;
Console.InputEncoding = System.Text.Encoding.UTF8;

Product product = new Product
{
    Id = 1,
    Name = "Notebook",
    Price = 2500,
    Quantity = 5
};
JsonSerializerOptions options = new JsonSerializerOptions { WriteIndented = true };
string jsonBefore = JsonSerializer.Serialize(product, options);
Console.WriteLine("До виключення Price:");
Console.WriteLine(jsonBefore);

string jsonAfter = JsonSerializer.Serialize(new
{
    product.Id,
    product.Name,
    product.Quantity
}, options);

Console.WriteLine("\nПісля виключення Price:");
Console.WriteLine(jsonAfter);



public class Product
{
    public int Id { get; set; }
    public string Name { get; set; } = "";
    public decimal Price { get; set; }
    public int Quantity { get; set; }
}
